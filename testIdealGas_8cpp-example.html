testIdealGas.cpp

Demonstrates kinetic Cauchy stress for an ideal gas with nonzero bulk velocity. The test verifies that MDStressLab subtracts the continuum velocity before forming the kinetic stress and writes the Cauchy-only momentum and mass density fields. It compares MethodSphere, MethodLdadConstant, and MethodLdadTrigonometric to the instantaneous ideal-gas pressure and to the expected mean mass density.

22{
23 Matrix3d meanStress= Matrix3d::Zero();
24 Matrix3d maxAbsDeviation= Matrix3d::Zero();
25 double meanMassDensity= 0.0;
26
27 for (int i_grid=0; i_grid<kineticStress.field.size(); ++i_grid)
28 {
29 const auto& stress= kineticStress.field[i_grid];
30 meanStress+= stress;
31 maxAbsDeviation= maxAbsDeviation.cwiseMax((stress-expectedStress).cwiseAbs());
32 meanMassDensity+= kineticStress.massDensityField[i_grid];
33
34 if (kineticStress.massDensityField[i_grid] < -epsilon)
35 MY_ERROR("Mass density should be nonnegative.");
36
37 if (kineticStress.massDensityField[i_grid] > epsilon)
38 {
39 Vector3d expectedVelocity=
40 kineticStress.momentumDensityField[i_grid]/kineticStress.massDensityField[i_grid];
41 if ((kineticStress.velocityField[i_grid]-expectedVelocity).norm() > 1e-12)
42 MY_ERROR("Continuum velocity does not equal momentum density divided by mass density.");
43 }
44 else
45 {
46 if (kineticStress.velocityField[i_grid].norm() > 1e-12)
47 MY_ERROR("Continuum velocity should be zero where mass density is zero.");
48 }
49 }
50
51 const double numberOfGridPoints= static_cast<double>(kineticStress.field.size());
52 meanStress/= numberOfGridPoints;
53 meanMassDensity/= numberOfGridPoints;
54
55 if ((rawVelocityStress-expectedStress).cwiseAbs().maxCoeff() < 10.0*meanStressTolerance)
56 MY_ERROR("Bulk velocity is too small to distinguish raw-velocity stress from relative-velocity stress.");
57
58 if (std::abs(meanMassDensity-expectedMassDensity) > densityTolerance)
59 {
60 std::cout << "Expected mass density = " << expectedMassDensity << " amu/A^3" << std::endl;
61 std::cout << "Mean computed mass density = " << meanMassDensity << " amu/A^3" << std::endl;
62 MY_ERROR("Ideal gas mass density mean does not match the normalized-kernel value.");
63 }
64
65 if ((meanStress-expectedStress).cwiseAbs().maxCoeff() > meanStressTolerance)
66 {
67 std::cout << "Expected stress:\n" << expectedStress << std::endl;
68 std::cout << "Mean computed stress:\n" << meanStress << std::endl;
69 MY_ERROR("Ideal gas kinetic stress mean does not match the instantaneous ideal gas value.");
70 }
71
72 if (maxAbsDeviation.diagonal().maxCoeff() > pointwiseStressTolerance)
73 {
74 std::cout << "Expected stress:\n" << expectedStress << std::endl;
75 std::cout << "Maximum pointwise absolute deviation:\n" << maxAbsDeviation << std::endl;
76 MY_ERROR("Ideal gas kinetic stress field is too far from the expected uniform field.");
77 }
78
79 std::cout << kineticStress.name << " mean mass density = "
